cmake_minimum_required(VERSION 3.24)
project(BO3SteamClient LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(T7 ${CMAKE_CURRENT_SOURCE_DIR}/../t7patch)
set(DEPS ${CMAKE_CURRENT_SOURCE_DIR}/../engine/deps)
file(GLOB_RECURSE CRYPT ${DEPS}/libtomcrypt/src/*.c)
list(FILTER CRYPT EXCLUDE REGEX "(_test\\.c|tab\\.c|/ocb3/)")
file(GLOB MATH ${DEPS}/libtommath/*.c)
add_library(community_crypto STATIC ${CRYPT} ${MATH})
target_include_directories(community_crypto PUBLIC ${DEPS}/libtomcrypt/src/headers ${DEPS}/libtommath)
target_compile_definitions(community_crypto PUBLIC LTM_DESC LTC_NO_FAST LTC_NO_RSA_BLINDING LTC_NO_FILE ARGTYPE=4 MP_NO_DEV_URANDOM PRIVATE LTC_SOURCE USE_LTM _CRT_SECURE_NO_WARNINGS)
file(GLOB PATCH ${T7}/*.cpp)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/generate_hooks.py ${T7}/Hooks.cpp ${CMAKE_CURRENT_BINARY_DIR}/CommunityHooks.cpp COMMAND_ERROR_IS_FATAL ANY)
list(FILTER PATCH EXCLUDE REGEX "/Hooks\\.cpp$")
add_library(BO3SteamFix SHARED native.cpp ${CMAKE_CURRENT_BINARY_DIR}/CommunityHooks.cpp ${PATCH})
target_include_directories(BO3SteamFix PRIVATE ${T7} ${DEPS}/rapidjson/include)
target_compile_definitions(BO3SteamFix PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX _CRT_SECURE_NO_WARNINGS T7PATCH_EXPORTS _WINDOWS _USRDLL)
target_compile_options(BO3SteamFix PRIVATE /FIvector /FIalgorithm /utf-8)
target_link_libraries(BO3SteamFix PRIVATE community_crypto ${T7}/minhook/lib/libMinHook.x64.lib ${T7}/detours/detours.lib ws2_32 shell32 psapi)
add_executable(BO3SteamAttach injector.cpp)
target_compile_definitions(BO3SteamAttach PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
target_link_libraries(BO3SteamAttach PRIVATE psapi shell32)
add_executable(ProtocolTests tests.cpp)
enable_testing()
add_test(NAME community_wire COMMAND ProtocolTests)
